\usepackage{fancyhdr} % Required for header and footer configuration
\usepackage{enumitem}

%----------------------------------------------------------------------------------------
%   TRACK SUBSECTION / SUBSUBSECTION NAMES FOR HEADERS
%----------------------------------------------------------------------------------------

\renewcommand{\subsectionmark}[1]{\markboth{#1}{}}
\renewcommand{\subsubsectionmark}[1]{\markright{#1}}

%----------------------------------------------------------------------------------------
%   PAGE HEADERS
%----------------------------------------------------------------------------------------

\fancypagestyle{plain}{%
    \fancyhf{}
    \renewcommand{\headrulewidth}{0pt}
    \fancyfoot[C]{\thepage}
}

\fancypagestyle{fancy}{%
    \fancyhf{}
    % Thin accent-colored rule below header
    \renewcommand{\headrule}{{\color{accent}\hrule width\headwidth height 0.4pt \vskip-0.4pt}}
    \renewcommand{\headrulewidth}{0.4pt}
    % Left: subsection name, Right: subsubsection name (truncated to fit)
    \fancyhead[L]{\small\headingfontfamily\color{fgalt}\leftmark}
    \fancyhead[R]{\parbox[b]{0.65\textwidth}{\raggedleft\small\headingfontfamily\color{fgalt}\rightmark}}
    % Page number centered at the bottom
    \fancyfoot[C]{\small\headingfontfamily\thepage}
}
